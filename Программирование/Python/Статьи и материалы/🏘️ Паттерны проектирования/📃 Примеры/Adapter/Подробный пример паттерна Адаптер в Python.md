# Подробный пример паттерна "Адаптер" в Python

Рассмотрим реальную ситуацию, где у нас есть две разные системы для работы с температурой: одна работает в градусах Цельсия, а другая в Фаренгейтах. Нам нужно сделать их совместимыми.

## Полный пример с объяснениями

```python
# ==================== 1. Определяем интерфейсы и несовместимые классы ====================

# Класс, который работает с температурой в Цельсиях (наш целевой интерфейс)
class CelsiusTemperature:
    def __init__(self, temperature: float):
        self.temperature = temperature
    
    def get_temperature(self) -> float:
        """Возвращает температуру в градусах Цельсия"""
        return self.temperature

# Класс, который работает с температурой в Фаренгейтах (адаптируемый класс)
class FahrenheitTemperature:
    def __init__(self, temperature: float):
        self.temperature = temperature
    
    def get_fahrenheit(self) -> float:
        """Возвращает температуру в градусах Фаренгейта"""
        return self.temperature

# ==================== 2. Создаем адаптер ====================

class FahrenheitToCelsiusAdapter(CelsiusTemperature):
    """
    Адаптер преобразует интерфейс FahrenheitTemperature к интерфейсу CelsiusTemperature.
    Используем композицию (объектный адаптер).
    """
    
    def __init__(self, fahrenheit_temp: FahrenheitTemperature):
        # Сохраняем ссылку на адаптируемый объект
        self.fahrenheit_temp = fahrenheit_temp
    
    def get_temperature(self) -> float:
        """
        Реализуем метод целевого интерфейса, выполняя преобразование:
        Формула: C = (F - 32) * 5/9
        """
        fahrenheit = self.fahrenheit_temp.get_fahrenheit()
        celsius = (fahrenheit - 32) * 5 / 9
        return round(celsius, 2)  # Округляем для красоты

# ==================== 3. Клиентский код, который работает с температурой ====================

def display_temperature(temp_source: CelsiusTemperature):
    """
    Функция ожидает объект с интерфейсом CelsiusTemperature.
    Она не знает, работает ли он напрямую с Цельсиями или через адаптер.
    """
    print(f"Текущая температура: {temp_source.get_temperature()}°C")

# ==================== 4. Использование ====================

if __name__ == "__main__":
    print("=== Работа с родной температурой в Цельсиях ===")
    celsius_temp = CelsiusTemperature(25.0)
    display_temperature(celsius_temp)  # Работает напрямую
    
    print("\n=== Попытка использовать температуру в Фаренгейтах напрямую ===")
    fahrenheit_temp = FahrenheitTemperature(77.0)  # 77°F = 25°C
    try:
        # Это вызовет ошибку, так как интерфейсы не совместимы
        display_temperature(fahrenheit_temp)
    except AttributeError as e:
        print(f"Ошибка: {e}")
        print("Нужен адаптер!")
    
    print("\n=== Использование адаптера ===")
    adapter = FahrenheitToCelsiusAdapter(fahrenheit_temp)
    display_temperature(adapter)  # Теперь работает через адаптер
    
    # Демонстрация с другими значениями
    print("\n=== Дополнительные примеры ===")
    for f in [32.0, 100.0, 212.0]:
        f_temp = FahrenheitTemperature(f)
        adapter = FahrenheitToCelsiusAdapter(f_temp)
        print(f"{f}°F = {adapter.get_temperature()}°C")
```

## Пошаговое объяснение кода:

1. **Определение интерфейсов и несовместимых классов**:
   - `CelsiusTemperature` - наш "целевой" интерфейс, с которым умеет работать клиентский код
   - `FahrenheitTemperature` - "чужой" интерфейс, который нужно адаптировать

2. **Создание адаптера**:
   - `FahrenheitToCelsiusAdapter` наследует от `CelsiusTemperature`, поэтому он совместим с целевым интерфейсом
   - В конструкторе принимает объект `FahrenheitTemperature` (композиция)
   - Реализует метод `get_temperature()`, выполняя преобразование Фаренгейты → Цельсии

3. **Клиентский код**:
   - Функция `display_temperature` работает только с объектами, имеющими метод `get_temperature()`
   - Она не знает о существовании Фаренгейтов и адаптеров

4. **Демонстрация работы**:
   - Показываем работу с родным интерфейсом (Цельсии)
   - Показываем, что прямое использование Фаренгейтов вызывает ошибку
   - Демонстрируем работу через адаптер
   - Приводим дополнительные примеры преобразования температур

## Вывод выполнения программы:

```
=== Использование адаптера ===
Текущая температура: 25.0°C

=== Дополнительные примеры ===
32.0°F = 0.0°C
100.0°F = 37.78°C
212.0°F = 100.0°C
```

## Ключевые моменты:

1. **Разделение ответственности**:
   - Адаптер знает как преобразовать один интерфейс в другой
   - Клиентский код не знает о деталях преобразования

2. **Гибкость**:
   - Мы можем добавить новые адаптеры для других шкал температуры (Кельвины и т.д.)
   - Не нужно изменять существующий клиентский код

3. **Соблюдение принципов SOLID**:
   - Принцип единственной ответственности (адаптер делает только преобразование)
   - Принцип открытости/закрытости (можно расширять без изменения существующего кода)

Этот пример показывает, как адаптер позволяет интегрировать несовместимые интерфейсы в вашу систему, не изменяя ни исходный код, ни клиентский код.
